% Custom commands

% #1 becomes bolded label. #2 is the descriptive text after. 
% Horizantal gap after bold text. 
% Slightly tightened vertical spacing between items
\newcommand{\resumeItem}[2]{
  \item{
    \textbf{#1}{\hspace{0.5mm}#2 \vspace{-0.5mm}}
  }
}

% Resume list item for "Position of Responsibility"
% Left aligned: Bold title #1 followed by short paragraph #2
% Right alined: showing secondary information such as dates or location #3 in small italic text.
\newcommand{\resumePOR}[3]{
\vspace{0.5mm}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
        \textbf{#1}\hspace{0.3mm}#2 & \textit{\small{#3}} 
    \end{tabular*}
    \vspace{-2mm}
}
 
% First Row: Left: Bold title #1. Right: small italic detail #4 for dates or location.
% Second Row: Left: small plain text #3 (position, degree, subtitle). Right:
% small plaintext detail #2 (city, gpa, or similar).
\newcommand{\resumeSubheading}[4]{
\vspace{0.5mm}\item
    \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
        \textbf{#1} & \textit{\footnotesize{#4}} \\
        \textit{\footnotesize{#3}} &  \footnotesize{#2}\\
    \end{tabular*}
    \vspace{-2.4mm}
}

% First Row: Left: Bold title #1. Right: small italic text for dates.
% Second Row: Left: small italic text for title/position. Right: small text for small detail.
\newcommand{\resumeProject}[4]{
\vspace{0.5mm}\item
    \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
        \textbf{#1} & \textit{\footnotesize{#3}} \\
        \footnotesize{\textit{#2}} & \footnotesize{#4}
    \end{tabular*}
    \vspace{-2.4mm}
}

% Remove vertical space from regular resumeItem
\newcommand{\resumeSubItem}[2]{\resumeItem{#1}{#2}\vspace{-4pt}}

\renewcommand{\labelitemii}{$\circ$}

% Replaces standard bullet with a tiny, vertically centered dot
% created using math-mode typesetting. For compact resume lists.
\renewcommand{\labelitemi}{$\vcenter{\hbox{\tiny$\bullet$}}$}

% leftmargin=*: full indentation according to document default
% labelsep=0mm: no space between the bullet and the text
\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=*,labelsep=0mm]}

% leftmargin=*: full indentation according to document default
% itemsep=1.7mm: adds a small vertical gap between items for readability,
% rightmargin=2ex: adds a little space on the right to keep the list tidy.
\newcommand{\resumeHeadingSkillStart}{\begin{itemize}[leftmargin=*,itemsep=1.7mm, rightmargin=2ex]}

% justify: LaTeX stretches spaces between words so that each line fills the full width.
% leftmargin=3ex: slight indentation on the left.
% rightmargin=2ex: small space on the right
% no itemsep & itemsep=0mm: no extra vertical spacing between items
% labelsep=1.2mm small gap between bullet and text.
\newcommand{\resumeItemListStart}{\begin{justify}\begin{itemize}[leftmargin=3ex, rightmargin=2ex, noitemsep,labelsep=1.2mm,itemsep=0mm]\small}


% ends the itemize environment for resume sub heading.
% adds 2mm of extra space below list for vertical separation.
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}\vspace{2mm}}

% closes itemize environment used for skill entries.
% adds -2mm of vertical space. Pulls next section closer for compact layout.
\newcommand{\resumeHeadingSkillEnd}{\end{itemize}\vspace{-2mm}}

% close itemize environment.
% close the surrounding justify block.
% adds -2mm of vertical space. Pulls next section closer for compact layout.
\newcommand{\resumeItemListEnd}{\end{itemize}\end{justify}\vspace{-2mm}}

% small vertical gap abover the section.
% open a tcoorbox to give heading background.
% Section title #1 is bold in tcolorbox. 
% adds -4mm of vertical space. Pulls next section closer for compact layout.
\newcommand{\cvsection}[1]{%
\vspace{2mm}
\begin{tcolorbox}
    \textbf{\large #1}
\end{tcolorbox}
    \vspace{-4mm}
}
